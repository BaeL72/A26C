STRINGS MANIPULATION:

Length: echo "${#str}" -> Print length of string
Substring Extraction: echo "${str:n}" -> Print from index n
                      echo "${str:0:n}" -> Print from 0 to n
                      echo "${str:(-n)}" -> Print last n chars
Find and Replace: echo "${str/old/new}" First match
                  echo "${str//old/new}" All matches
                  echo "${str/#old/new}" At start
                  echo "${str/%old/new}" At end
Remove Prefix/Suffix: file="hello.txt"
                      echo "${file%.txt}" -> hello Remove shortest match
                      echo "${file%%.*}" -> hello Remove longest match
                      echo "${file#hello}" -> .txt Remove shortest match
                      echo "${file##*.}" -> txt Remove longest match
Case Conversion: echo "${string^^}" All uppercase
                 echo "${string,,}" All lowercase
                 echo "${string^}" Capitalize first char
                 echo "${string,}" Lowercase first char

echo "hello" | rev -> "olleh" Reverse string
echo "hello" | tr -cd 'l' | wc -c -> 2 Count occurrences of a char
echo "h-e-l-l-o" | tr -d '-' -> "hello" Remove specific chars 
echo "hello" | tr 'a-z' 'A-Z' -> "HELLO" Replace chars

LOOPS:

SYNTAX:

FOR LOOP
for item in apple banana cherry; do
  echo "$item"
done

C-STYLE LOOP
for ((i=0; i<5; i++)); do
  echo "$i"
done

LOOP OVER AN ARRAY
fruits=("apple" "banana" "cherry")
for fruit in "${fruits[@]}"; do
  echo "$fruit"
done

LOOP OVER FILES
for file in /etc/*.conf; do
  echo "$file"
done

LOOP OVER A RANGE:
for i in {1..5}; do
  echo "$i"
done

WITH STEP:
for i in {0..20..5}; do
  echo "$i"
done

WHILE LOOP:

  echo "$1"
  ((i++))
done

TRAP:

trap runs a command when the script receives a signal or exits.

SIGNALS:

EXIT Script exits(any reason)
INT Ctrl + C
TERM Kill command
ERR Any command fails
DEBUG Before any command
RETURN Function returns

EXAMPLE:

trap 'rm -f "$tmpfile"' EXIT Always runs on exit

trap 'echo "Interrupted!"; exit' INT TERM Multiple signals
